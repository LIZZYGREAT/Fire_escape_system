    │   └── style.css               # 样式文件 (深色大屏 UI 设计，保证 Canvas 满屏)

    │
    └── js/                         # 模块化前端引擎 (严格采用 ES6 Modules 划分模块)
        ├── main.js                 # 前端总控：双轨时钟引擎 (Logic Tick 与 Render Tick)
        ├── network.js              # 通信模块：专职负责 WebSocket 握手、断线重连与 Payload 解析
        ├── renderer.js             # 渲染管线：封装 Canvas API，管理拓扑层、火势层、硬件状态层
        └── particles.js            # 物理引擎：火场混沌粒子与逃生流线粒子的运动学推算
